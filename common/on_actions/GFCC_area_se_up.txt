on_actions = {
    #每周触发的地区系统
    on_weekly = {
        effect = {
            if = {
                limit = {
                    is_ai = no
                }
                every_state = {
                    #污染度设定
                    if = {
                        limit = {
                            GFCC_area_Pollution_B1 > 0
                        }
                        GFCC_area_Pollution_effects_1 = yes #净化塔检测
                    }
                    GFCC_area_Pollution_effects_2 = yes #黑区周边辐射
                    GFCC_area_Pollution_effects_4 = yes #地区BUFF效果修正
                    GFCC_area_Pollution_effects_5 = yes #净化区、白区效果辐射
                    GFCC_area_Pollution_effects_6 = yes #黄区辐射效果
                    GFCC_area_Pollution_effects_7 = yes #红区辐射效果

                    #安全度与重建度设定
                    GFCC_area_security_effects_2 = yes #安全度每周更新
                    # GFCC_area_reconstruction_effects_1 = yes #重建度受战争影响
                    GFCC_area_reconstruction_effects_3 = yes #重建度增长（受安全度影响）
                    GFCC_area_reconstruction_effects_5 = yes #污染度对重建度效果修正
                    GFCC_area_security_effects_3 = yes #安全度效果修正
                    GFCC_area_reconstruction_effects_4 = yes #重建度效果修正
                }
            }
        }
    }
    on_daily = {
        effect = {
            every_state = {
                GFCC_area_reconstruction_effects_1 = yes #重建度受战争影响
            }
        }
    }
    on_startup = {
        effect = {
            GER = {
                if = {
                    limit = { is_ai = no NOT = { has_country_flag = APDC_factory_diagnostics_disabled } }
                    set_temp_variable = { APDC_fd_stage = 0 }
                    APDC_log_factory_snapshot = yes
                }
            }
            every_state = {
                GFCC_area_Pollution_effects_3 = yes #所有省份设定初始值 坍塌污染初始化
                GFCC_area_Pollution_effects_4 = yes #地区BUFF效果修正
                GFCC_area_security_effects_0 = yes #安全度初始化
                GFCC_area_reconstruction_effects_0 = yes #重建度初始化
                GFCC_area_security_effects_3 = yes #安全度效果修正
                GFCC_area_reconstruction_effects_4 = yes #重建度效果修正
            }

            GER = {
                if = {
                    limit = { is_ai = no NOT = { has_country_flag = APDC_factory_diagnostics_disabled } }
                    set_temp_variable = { APDC_fd_stage = 1 }
                    APDC_log_factory_snapshot = yes
                }
            }

            #GUI窗口测试
            every_country = {
                set_country_flag = GFCC_area_gui_hide_window #默认隐藏地区窗口
            }
        }
    }
}
