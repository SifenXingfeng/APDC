# Generic cooperation-organization container API.
#
# A container is anchored to a country scope. The anchor stores the member
# list, chair, legislation, ballot and queue data. A member country stores
# only APDC_cooperation_organization_container, a scoped variable pointing to
# that anchor. Containers are independent from military factions, and a
# country can belong to at most one container.
#
# Create a new container on its anchor country:
#   set_variable = { APDC_cooperation_organization_definition_id = 3 } # unique id
#   set_variable = { APDC_cooperation_organization_name_id = 3 }
#   set_variable = { APDC_cooperation_organization_icon_id = 3 }
#   APDC_create_cooperation_organization = yes
# Every container automatically receives the shared placeholder legislation.
#
# Add a country to an existing container:
#   set_variable = { APDC_cooperation_organization_join_target = SOV }
#   APDC_join_cooperation_organization = yes
#
# Remove the current country from its container:
#   APDC_leave_cooperation_organization = yes
# Alias names intended for country-history/event code:
#   APDC_add_country_to_cooperation_organization = yes
#   APDC_remove_country_from_cooperation_organization = yes
#
# Dissolve a container from its anchor scope:
#   APDC_remove_cooperation_organization = yes

# Build the compact diplomacy list in the currently viewed country scope.  The
# chair is inserted first and omitted from the ordinary-member pass.  Keeping
# both this array and the open flag local to the selected country prevents a
# panel opened for one country from leaking into another country's view.
APDC_sync_diplomacy_cooperation_organization_member_view = {
	clear_array = APDC_diplomacy_cooperation_organization_display_members
	clear_variable = APDC_diplomacy_cooperation_organization_panel_open
	if = {
		limit = { APDC_has_cooperation_organization = yes }
		set_variable = { global.APDC_diplomacy_cooperation_sync_member = THIS }
		for_each_scope_loop = {
			array = APDC_cooperation_organization_display_chair
			var:global.APDC_diplomacy_cooperation_sync_member = {
				add_to_array = {
					array = APDC_diplomacy_cooperation_organization_display_members
					value = PREV
				}
			}
		}
		for_each_scope_loop = {
			array = APDC_cooperation_organization_display_members
			if = {
				limit = { APDC_is_current_cooperation_organization_chair = no }
				var:global.APDC_diplomacy_cooperation_sync_member = {
					add_to_array = {
						array = APDC_diplomacy_cooperation_organization_display_members
						value = PREV
					}
				}
			}
		}
		clear_variable = global.APDC_diplomacy_cooperation_sync_member
		set_variable = { APDC_diplomacy_cooperation_organization_panel_open = 1 }
	}
}

# Run in a member country scope. All GUI arrays are read-only local mirrors;
# authoritative organization state remains inside the anchor container.
APDC_sync_cooperation_organization_member_view = {
	APDC_clear_cooperation_organization_member_view_data = yes

		limit = { APDC_has_cooperation_organization = yes }

			for_each_scope_loop = {
				array = APDC_cooperation_organization_container_chair
				var:global.APDC_cooperation_sync_member = {
					add_to_array = {
						array = APDC_cooperation_organization_display_chair
						value = PREV
					}
				}

				array = APDC_cooperation_organization_container_members
				var:global.APDC_cooperation_sync_member = {
					add_to_array = {
						array = APDC_cooperation_organization_display_members

					}

			}

			var:global.APDC_cooperation_sync_member = {
				set_variable = { APDC_cooperation_vote_state = PREV.APDC_cooperation_container_vote_state }
				set_variable = { APDC_cooperation_vote_days_remaining = PREV.APDC_cooperation_container_vote_days_remaining }
				set_variable = { APDC_cooperation_active_law_group = PREV.APDC_cooperation_container_active_law_group }
				set_variable = { APDC_cooperation_active_law_option = PREV.APDC_cooperation_container_active_law_option }
				set_variable = { APDC_cooperation_active_law_proposer = PREV.APDC_cooperation_container_active_law_proposer }
			}

			# A for_each_loop value is a temporary numeric variable owned by the
			# current execution scope. It cannot be read reliably after switching
			# from the container to a member country. Copy it to a persistent global
			# numeric bridge before the switch, then append in the same order. This
			# remains fully data-driven: no group id or schema size is enumerated.
			for_each_loop = {
				array = APDC_cooperation_container_law_groups
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}
			}
			for_each_loop = {
				array = APDC_cooperation_container_law_tab_1_groups
				value = v

					add_to_array = { array = APDC_cooperation_law_tab_1_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}

			for_each_loop = {

					add_to_array = { array = APDC_cooperation_law_tab_2_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}
			}
			for_each_loop = {
				array = APDC_cooperation_container_law_tab_3_groups
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_tab_3_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}
			}
			for_each_loop = {
				array = APDC_cooperation_container_law_tab_4_groups
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_tab_4_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}
			}
			for_each_loop = {
				array = APDC_cooperation_container_law_tab_5_groups
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_tab_5_groups value = var:global.APDC_cooperation_sync_numeric_value }
				}
			}
			for_each_loop = {
				array = APDC_cooperation_container_law_group_tabs
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_group_tabs value = var:global.APDC_cooperation_sync_numeric_value }
				}

			for_each_loop = {
				array = APDC_cooperation_container_law_option_counts
				value = v
				set_variable = { global.APDC_cooperation_sync_numeric_value = v }
				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_option_counts value = var:global.APDC_cooperation_sync_numeric_value }
				}

			for_each_loop = {

				array = APDC_cooperation_container_law_option_3_available

			for_each_loop = {

				value = v

			for_each_loop = {

					add_to_array = { array = APDC_cooperation_law_queue_groups value = var:global.APDC_cooperation_sync_numeric_value }

				array = APDC_cooperation_container_law_queue_options
				value = v

				var:global.APDC_cooperation_sync_member = {
					add_to_array = { array = APDC_cooperation_law_queue_options value = var:global.APDC_cooperation_sync_numeric_value }

APDC_create_cooperation_organization = {
	if = {
		limit = {
			has_variable = APDC_cooperation_organization_definition_id
			NOT = { has_country_flag = APDC_cooperation_organization_container }
		}
		set_country_flag = APDC_cooperation_organization_container
		# Removed v2 switch: every container always owns legislation state.
		clear_variable = APDC_cooperation_organization_has_legislation
		clear_array = APDC_cooperation_organization_container_chair
		clear_array = APDC_cooperation_organization_container_members
		clear_array = APDC_cooperation_organization_template_opening_members
		if = {
			limit = { NOT = { has_variable = APDC_cooperation_organization_name_id } }
			set_variable = { APDC_cooperation_organization_name_id = 0 }
		}
		if = {
			limit = { NOT = { has_variable = APDC_cooperation_organization_icon_id } }
			set_variable = { APDC_cooperation_organization_icon_id = 0 }
		}
		if = {
			limit = {
				NOT = { is_in_array = { array = global.APDC_cooperation_organization_containers value = THIS } }
			}
			add_to_array = {
				array = global.APDC_cooperation_organization_containers
				value = THIS
			}
		}
	}
	else_if = {
		limit = {
			has_country_flag = APDC_cooperation_organization_container
			NOT = { is_in_array = { array = global.APDC_cooperation_organization_containers value = THIS } }
		}
		add_to_array = { array = global.APDC_cooperation_organization_containers value = THIS }
	}
	if = {
		limit = { has_country_flag = APDC_cooperation_organization_container }
		APDC_initialize_cooperation_legislation_container = yes
	}
}

APDC_remove_cooperation_organization = {
	if = {
		limit = { has_country_flag = APDC_cooperation_organization_container }
		for_each_scope_loop = {
			array = APDC_cooperation_organization_container_members
			clear_variable = APDC_cooperation_organization_container
			clr_country_flag = APDC_cooperation_organization_fifth_international_member
			clr_country_flag = APDC_cooperation_organization_international_rossartrism_member
			APDC_clear_cooperation_organization_member_view_data = yes
			clear_variable = APDC_cooperation_organization_selected_tab
		}
		remove_from_array = { array = global.APDC_cooperation_organization_containers value = THIS }
		clear_array = APDC_cooperation_organization_container_chair
		clear_array = APDC_cooperation_organization_container_members
		APDC_clear_cooperation_legislation_container_data = yes
		clear_array = APDC_cooperation_organization_template_opening_members
		clear_variable = APDC_cooperation_organization_definition_id
		clear_variable = APDC_cooperation_organization_name_id
		clear_variable = APDC_cooperation_organization_icon_id
		clear_variable = APDC_cooperation_organization_has_legislation
		clr_country_flag = APDC_cooperation_organization_container
	}
}

APDC_clear_cooperation_organization_membership = {
	if = {
		limit = { APDC_has_cooperation_organization = yes }
		var:APDC_cooperation_organization_container = {
			remove_from_array = { array = APDC_cooperation_organization_container_members value = PREV }
			remove_from_array = { array = APDC_cooperation_organization_container_chair value = PREV }
			remove_from_array = { array = APDC_cooperation_container_vote_supporters value = PREV }
			remove_from_array = { array = APDC_cooperation_container_vote_abstainers value = PREV }
			remove_from_array = { array = APDC_cooperation_container_vote_vetoers value = PREV }
			APDC_sync_cooperation_organization_all_member_views = yes
		}
		clear_variable = APDC_cooperation_organization_container
	}
	clr_country_flag = APDC_cooperation_organization_fifth_international_member
	clr_country_flag = APDC_cooperation_organization_international_rossartrism_member
	APDC_clear_cooperation_organization_member_view_data = yes
	clear_variable = APDC_cooperation_organization_selected_tab
}

APDC_join_cooperation_organization = {
	if = {
		limit = {
			has_variable = APDC_cooperation_organization_join_target
			var:APDC_cooperation_organization_join_target = {
				has_country_flag = APDC_cooperation_organization_container
			}
		}
		APDC_clear_cooperation_organization_membership = yes
		var:APDC_cooperation_organization_join_target = {
			add_to_array = { array = APDC_cooperation_organization_container_members value = PREV }
			PREV = { set_variable = { APDC_cooperation_organization_container = PREV } }

			if = {
				limit = { check_variable = { APDC_cooperation_organization_definition_id = 1 } }
				PREV = { set_country_flag = APDC_cooperation_organization_fifth_international_member }
			}
			else_if = {
				limit = { check_variable = { APDC_cooperation_organization_definition_id = 2 } }
				PREV = { set_country_flag = APDC_cooperation_organization_international_rossartrism_member }
			}

			if = {
				limit = { check_variable = { APDC_cooperation_container_vote_state = 1 } }
				add_to_array = { array = APDC_cooperation_container_vote_abstainers value = PREV }
			}
			APDC_sync_cooperation_organization_all_member_views = yes
		}
	}
	clear_variable = APDC_cooperation_organization_join_target
}

APDC_leave_cooperation_organization = {
	APDC_clear_cooperation_organization_membership = yes
}

# Run in the new chair country's scope. The country must already be a member.
APDC_set_cooperation_organization_chair = {
	if = {
		limit = { APDC_has_cooperation_organization = yes }
		var:APDC_cooperation_organization_container = {
			clear_array = APDC_cooperation_organization_container_chair
			add_to_array = { array = APDC_cooperation_organization_container_chair value = PREV }
			APDC_sync_cooperation_organization_all_member_views = yes
		}
	}
}

# Rebuild a container's member array from the authoritative scoped variable.
# Run in the container anchor scope.
APDC_refresh_cooperation_organization_container = {
	if = {
		limit = { has_country_flag = APDC_cooperation_organization_container }
		clear_array = APDC_cooperation_organization_container_members
		set_variable = { global.APDC_cooperation_refresh_container = THIS }
		every_country = {
			limit = {
				has_variable = APDC_cooperation_organization_container
				var:APDC_cooperation_organization_container = { tag = var:global.APDC_cooperation_refresh_container }
			}
			var:global.APDC_cooperation_refresh_container = {
				add_to_array = { array = APDC_cooperation_organization_container_members value = PREV }
			}
		}
		clear_variable = global.APDC_cooperation_refresh_container
		APDC_sync_cooperation_organization_all_member_views = yes
	}
}

		SOV = { PREV = { set_variable = { APDC_cooperation_organization_join_target = PREV } } }
		APDC_join_cooperation_organization = yes
	}
}

		}

	if = {

# Rebuild the registry and migrate saves made by the old flag-based system.
APDC_initialize_cooperation_organization_containers = {
	clear_array = global.APDC_cooperation_organization_containers
	every_country = {
		limit = { has_country_flag = APDC_cooperation_organization_container }
		add_to_array = { array = global.APDC_cooperation_organization_containers value = THIS }
	}

	every_country = {
		if = {
			limit = {
				has_country_flag = APDC_cooperation_organization_fifth_international_member
				NOT = { has_variable = APDC_cooperation_organization_container }
			}
			APDC_join_fifth_international_cooperation_organization = yes
		}
		else_if = {
			limit = {
				has_country_flag = APDC_cooperation_organization_international_rossartrism_member
				NOT = { has_variable = APDC_cooperation_organization_container }
			}
			APDC_join_international_rossartrism_cooperation_organization = yes
		}
	}

	for_each_scope_loop = {
		array = global.APDC_cooperation_organization_containers
		# Upgrade containers from the earlier definition-id-only format.
		if = {
			limit = { check_variable = { APDC_cooperation_organization_definition_id = 1 } }
			set_variable = { APDC_cooperation_organization_name_id = 1 }
			set_variable = { APDC_cooperation_organization_icon_id = 1 }
		}
		else_if = {
			limit = { check_variable = { APDC_cooperation_organization_definition_id = 2 } }
			set_variable = { APDC_cooperation_organization_name_id = 2 }
			set_variable = { APDC_cooperation_organization_icon_id = 2 }
		}
		clear_variable = APDC_cooperation_organization_has_legislation
		APDC_refresh_cooperation_organization_container = yes
		APDC_initialize_cooperation_legislation_container = yes
		APDC_repair_cooperation_timer_daily_rate_container = yes
	}
}
