set_temp_variable = { apdc_cw_largest = apdc_cw_political^apdc_cw_j }
set_temp_variable = { apdc_cw_remainder_index = apdc_cw_j }
} } }
set_variable = { apdc_cw_political^apdc_cw_source_index = apdc_cw_new }
set_temp_variable = { apdc_cw_rounding = 100 }
subtract_from_temp_variable = { apdc_cw_rounding = apdc_cw_total }
add_to_variable = { apdc_cw_political^apdc_cw_remainder_index = apdc_cw_rounding }
}
