# Required temporary definition fields:
#   APDC_cooperation_new_law_group_id             globally unique numeric id
#   APDC_cooperation_new_law_group_tab            1..5
#   APDC_cooperation_new_law_group_option_count   1..4
#   APDC_cooperation_new_law_group_default_option 1..option_count
# Run on the organization anchor. This is the only schema-writing path used by
# both the default placeholder package and future organization-specific laws.
APDC_register_cooperation_law_group_container = {
	if = {
		limit = {
			has_country_flag = APDC_cooperation_organization_container
			check_variable = { APDC_cooperation_new_law_group_id > 0 }
			check_variable = { APDC_cooperation_new_law_group_tab > 0 }
			check_variable = { APDC_cooperation_new_law_group_tab < 6 }
			check_variable = { APDC_cooperation_new_law_group_option_count > 0 }
			check_variable = { APDC_cooperation_new_law_group_option_count < 5 }
			OR = {
				check_variable = { APDC_cooperation_new_law_group_default_option = 1 }
				AND = { check_variable = { APDC_cooperation_new_law_group_default_option = 2 } check_variable = { APDC_cooperation_new_law_group_option_count > 1 } }
				AND = { check_variable = { APDC_cooperation_new_law_group_default_option = 3 } check_variable = { APDC_cooperation_new_law_group_option_count > 2 } }
				AND = { check_variable = { APDC_cooperation_new_law_group_default_option = 4 } check_variable = { APDC_cooperation_new_law_group_option_count > 3 } }
			}
		}

		set_temp_variable = { APDC_cooperation_law_schema_target_size = APDC_cooperation_new_law_group_id }
		add_to_temp_variable = { APDC_cooperation_law_schema_target_size = 1 }
		while_loop_effect = {
			limit = { check_variable = { APDC_cooperation_container_law_option_counts^num < APDC_cooperation_law_schema_target_size } }
			add_to_array = { array = APDC_cooperation_container_law_group_tabs value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_option_counts value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_selected_options value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_group_locks value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_option_1_available value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_option_2_available value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_option_3_available value = 0 }
			add_to_array = { array = APDC_cooperation_container_law_option_4_available value = 0 }
		}

		if = {
			limit = { NOT = { is_in_array = { array = APDC_cooperation_container_law_groups value = var:APDC_cooperation_new_law_group_id } } }
			add_to_array = { array = APDC_cooperation_container_law_groups value = var:APDC_cooperation_new_law_group_id }
		}
		# Re-registering an existing group may move it to another tab. HOI4 logs
		# an invalid-index error when remove_from_array is called for a missing
		# numeric value, so test each tab before removing it.
		if = { limit = { is_in_array = { array = APDC_cooperation_container_law_tab_1_groups value = var:APDC_cooperation_new_law_group_id } } remove_from_array = { array = APDC_cooperation_container_law_tab_1_groups value = var:APDC_cooperation_new_law_group_id } }
		if = { limit = { is_in_array = { array = APDC_cooperation_container_law_tab_2_groups value = var:APDC_cooperation_new_law_group_id } } remove_from_array = { array = APDC_cooperation_container_law_tab_2_groups value = var:APDC_cooperation_new_law_group_id } }
		if = { limit = { is_in_array = { array = APDC_cooperation_container_law_tab_3_groups value = var:APDC_cooperation_new_law_group_id } } remove_from_array = { array = APDC_cooperation_container_law_tab_3_groups value = var:APDC_cooperation_new_law_group_id } }
		if = { limit = { is_in_array = { array = APDC_cooperation_container_law_tab_4_groups value = var:APDC_cooperation_new_law_group_id } } remove_from_array = { array = APDC_cooperation_container_law_tab_4_groups value = var:APDC_cooperation_new_law_group_id } }
		if = { limit = { is_in_array = { array = APDC_cooperation_container_law_tab_5_groups value = var:APDC_cooperation_new_law_group_id } } remove_from_array = { array = APDC_cooperation_container_law_tab_5_groups value = var:APDC_cooperation_new_law_group_id } }
		if = { limit = { check_variable = { APDC_cooperation_new_law_group_tab = 1 } } add_to_array = { array = APDC_cooperation_container_law_tab_1_groups value = var:APDC_cooperation_new_law_group_id } }
		else_if = { limit = { check_variable = { APDC_cooperation_new_law_group_tab = 2 } } add_to_array = { array = APDC_cooperation_container_law_tab_2_groups value = var:APDC_cooperation_new_law_group_id } }
		else_if = { limit = { check_variable = { APDC_cooperation_new_law_group_tab = 3 } } add_to_array = { array = APDC_cooperation_container_law_tab_3_groups value = var:APDC_cooperation_new_law_group_id } }
		else_if = { limit = { check_variable = { APDC_cooperation_new_law_group_tab = 4 } } add_to_array = { array = APDC_cooperation_container_law_tab_4_groups value = var:APDC_cooperation_new_law_group_id } }
		else = { add_to_array = { array = APDC_cooperation_container_law_tab_5_groups value = var:APDC_cooperation_new_law_group_id } }

		set_variable = { APDC_cooperation_container_law_group_tabs^APDC_cooperation_new_law_group_id = APDC_cooperation_new_law_group_tab }
		set_variable = { APDC_cooperation_container_law_option_counts^APDC_cooperation_new_law_group_id = APDC_cooperation_new_law_group_option_count }
		set_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_new_law_group_id = APDC_cooperation_new_law_group_default_option }
		set_variable = { APDC_cooperation_container_law_group_locks^APDC_cooperation_new_law_group_id = 0 }
		set_variable = { APDC_cooperation_container_law_option_1_available^APDC_cooperation_new_law_group_id = 1 }
		set_variable = { APDC_cooperation_container_law_option_2_available^APDC_cooperation_new_law_group_id = 0 }
		set_variable = { APDC_cooperation_container_law_option_3_available^APDC_cooperation_new_law_group_id = 0 }
		set_variable = { APDC_cooperation_container_law_option_4_available^APDC_cooperation_new_law_group_id = 0 }
		if = { limit = { check_variable = { APDC_cooperation_new_law_group_option_count > 1 } } set_variable = { APDC_cooperation_container_law_option_2_available^APDC_cooperation_new_law_group_id = 1 } }
		if = { limit = { check_variable = { APDC_cooperation_new_law_group_option_count > 2 } } set_variable = { APDC_cooperation_container_law_option_3_available^APDC_cooperation_new_law_group_id = 1 } }
		if = { limit = { check_variable = { APDC_cooperation_new_law_group_option_count > 3 } } set_variable = { APDC_cooperation_container_law_option_4_available^APDC_cooperation_new_law_group_id = 1 } }
	}
}

# APDC_cooperation_law_group_id is supplied by the dynamic law-group entry;
# APDC_cooperation_submitted_option is set by the clicked option button.
APDC_submit_cooperation_law_container = {
apdc_nr_refresh_rules = yes

	if = {
		limit = {
			has_country_flag = APDC_cooperation_organization_container
			is_in_array = { array = APDC_cooperation_container_law_groups value = var:APDC_cooperation_submitted_group }
check_variable = { apdc_nr_visibles^APDC_cooperation_submitted_group = 1 }
			# Validate every legal option count explicitly for this game build.
			OR = {
				check_variable = { APDC_cooperation_submitted_option = 1 }
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 2 }
					check_variable = { APDC_cooperation_container_law_option_counts^APDC_cooperation_submitted_group > 1 }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 3 }
					check_variable = { APDC_cooperation_container_law_option_counts^APDC_cooperation_submitted_group > 2 }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 4 }
					check_variable = { APDC_cooperation_container_law_option_counts^APDC_cooperation_submitted_group > 3 }
				}
			}
			check_variable = { APDC_cooperation_container_law_group_locks^APDC_cooperation_submitted_group = 0 }
			OR = {
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 1 }
					NOT = { check_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_submitted_group = 1 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 2 }
					NOT = { check_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_submitted_group = 2 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 3 }
					NOT = { check_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_submitted_group = 3 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 4 }
					NOT = { check_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_submitted_group = 4 } }
				}
			}
			OR = {
				AND = { check_variable = { APDC_cooperation_submitted_option = 1 } check_variable = { APDC_cooperation_container_law_option_1_available^APDC_cooperation_submitted_group = 1 } }
				AND = { check_variable = { APDC_cooperation_submitted_option = 2 } check_variable = { APDC_cooperation_container_law_option_2_available^APDC_cooperation_submitted_group = 1 } }

APDC_resolve_cooperation_law_vote_container = {
apdc_nr_ensure = yes
set_variable = { apdc_nr_transaction_id = APDC_cooperation_container_active_law_group }
set_variable = { apdc_nr_old_option = APDC_cooperation_container_law_selected_options^APDC_cooperation_container_active_law_group }
set_variable = { apdc_nr_new_option = APDC_cooperation_container_active_law_option }

	# Copy the active identifiers to local indices before mutating the container
	# arrays. Clausewitz reliably accepts temporary variables as array indices.
	set_temp_variable = { APDC_cooperation_resolved_group = APDC_cooperation_container_active_law_group }
	set_temp_variable = { APDC_cooperation_resolved_option = APDC_cooperation_container_active_law_option }
	if = {
		limit = { check_variable = { APDC_cooperation_container_vote_supporters^num > APDC_cooperation_container_vote_vetoers^num } }
		if = {
limit = { check_variable = { apdc_nr_natives^apdc_nr_transaction_id = 1 } }
apdc_nr_repeal = yes
}
set_variable = { APDC_cooperation_container_law_selected_options^APDC_cooperation_resolved_group = APDC_cooperation_resolved_option }
if = {
limit = { check_variable = { apdc_nr_natives^apdc_nr_transaction_id = 1 } }
apdc_nr_enact = yes
}

		add_to_variable = { APDC_cooperation_container_enacted_law_revision = 1 }
		for_each_scope_loop = {
			array = APDC_cooperation_organization_container_members
			set_variable = { APDC_cooperation_last_enacted_law_group = PREV.APDC_cooperation_container_active_law_group }
			set_variable = { APDC_cooperation_last_enacted_law_option = PREV.APDC_cooperation_container_active_law_option }
			set_variable = { APDC_cooperation_law_effect_revision = PREV.APDC_cooperation_container_enacted_law_revision }
			set_country_flag = APDC_cooperation_law_effects_active
		}
	}

else = {
if = {
limit = { check_variable = { apdc_nr_natives^apdc_nr_transaction_id = 1 } }
meta_effect = {
text = {
apdc_cooperation_rule_[ID]_on_fail = yes
}
ID = "[?apdc_nr_transaction_id|0..]"
}
}
}

	# Passing or failing ends this group's lock; another option may be
	# proposed after the mandatory recess.
	set_variable = { APDC_cooperation_container_law_group_locks^APDC_cooperation_resolved_group = 0 }
	clear_array = APDC_cooperation_container_vote_supporters
	clear_array = APDC_cooperation_container_vote_abstainers
	clear_array = APDC_cooperation_container_vote_vetoers
	clear_variable = APDC_cooperation_container_active_law_group
	clear_variable = APDC_cooperation_container_active_law_option
	clear_variable = APDC_cooperation_container_active_law_proposer
	set_variable = { APDC_cooperation_container_vote_state = 2 }
	set_variable = { APDC_cooperation_container_vote_days_remaining = 30 }
	APDC_sync_cooperation_organization_all_member_views = yes
}


APDC_cooperation_legislation_daily_tick_container = {
	if = {
		limit = { has_country_flag = APDC_cooperation_legislation_container_initialized_v8 }
		if = {
			limit = { check_variable = { APDC_cooperation_container_vote_state = 1 } }
			add_to_variable = { APDC_cooperation_container_vote_days_remaining = -1 }
			if = {
				limit = { check_variable = { APDC_cooperation_container_vote_days_remaining < 1 } }
				APDC_resolve_cooperation_law_vote_container = yes
			}
		}
		else_if = {
			limit = { check_variable = { APDC_cooperation_container_vote_state = 2 } }
			add_to_variable = { APDC_cooperation_container_vote_days_remaining = -1 }
			if = {
				limit = { check_variable = { APDC_cooperation_container_vote_days_remaining < 1 } }
				set_variable = { APDC_cooperation_container_vote_state = 0 }
				clear_variable = APDC_cooperation_container_vote_days_remaining
				APDC_start_next_cooperation_law_vote_container = yes
			}
		}
		else = {
			APDC_start_next_cooperation_law_vote_container = yes
		}
		APDC_sync_cooperation_organization_all_member_scalar_views = yes
	}
}
