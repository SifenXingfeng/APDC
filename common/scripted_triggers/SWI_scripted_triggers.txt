###### AI Triggers ##### 

SWI_is_historical_ai = { 
	is_ai = yes 
	OR = {
		AND = {
			is_historical_focus_on = yes 
			has_game_rule = {
				rule = SWI_AI_BEHAVIOR
				option = DEFAULT
			}
		}
		has_country_flag = SWI_AI_RANDOM_HISTORICAL
		has_game_rule = {
			rule = SWI_ai_behavior
			option = HISTORICAL
		}
	}	
}
