on_actions = {
	# Root initialization for the system: only countries without an explicitly
	# assigned economic system are classified here.
	on_startup = {
		effect = {
			every_country = {
				limit = { NOT = { APDC_has_economic_system = yes } }
				APDC_assign_initial_economic_system = yes
			}
		}
	}

	# Released countries and other dynamically created tags do not necessarily
	# pass through a history file.  This fills only a missing system and never
	# recalibrates or replaces an existing choice.
	on_monthly = {
		effect = {
			every_country = {
				limit = { NOT = { APDC_has_economic_system = yes } }
				APDC_assign_initial_economic_system = yes
			}
		}
	}
}
