# APDC cabinet member system
# IDs 1/2 are real German characters. IDs 3-47 are five Germany-only
# visual placeholders per ideology and deliberately grant no modifiers.

APDC_cabinet_clear_slot_1 = {
	if = { limit = { check_variable = { APDC_cabinet_slot_1 = 1 } } remove_ideas = GER_APDC_romy_riefenstahl }
	else_if = { limit = { check_variable = { APDC_cabinet_slot_1 = 2 } } remove_ideas = GER_APDC_eugene_rugosa }
	clear_variable = APDC_cabinet_slot_1
}

}

APDC_cabinet_refresh_selected_candidate = {
	clear_variable = APDC_cabinet_selected_candidate
	if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 1 } has_variable = APDC_cabinet_slot_1 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_1 }
	}
	else_if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 2 } has_variable = APDC_cabinet_slot_2 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_2 }
	}
	else_if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 3 } has_variable = APDC_cabinet_slot_3 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_3 }
	}
	else_if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 4 } has_variable = APDC_cabinet_slot_4 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_4 }
	}
	else_if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 5 } has_variable = APDC_cabinet_slot_5 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_5 }
	}
	else_if = {
		limit = { check_variable = { APDC_cabinet_selected_slot = 6 } has_variable = APDC_cabinet_slot_6 }
		set_variable = { APDC_cabinet_selected_candidate = APDC_cabinet_slot_6 }
	}
}

		# A party is in the top-six band when fewer than six parties outrank it.
		# The band can contain more than six parties only when the cutoff is tied.
		set_variable = { APDC_cabinet_top_six_candidate_count = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_communism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_socialism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_social_democracy = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_conservatism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_liberalism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_authoritarianism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_dictatorship = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_fascism = 0 }
		set_variable = { APDC_cabinet_top_six_candidate_fundamentalism = 0 }
		if = { limit = { APDC_is_communism_in_governing_coalition = yes check_variable = { APDC_cabinet_higher_influence_count_communism < 6 } } set_variable = { APDC_cabinet_top_six_candidate_communism = 1 } add_to_variable = { APDC_cabinet_top_six_candidate_count = 1 } }
		if = { limit = { APDC_is_socialism_in_governing_coalition = yes check_variable = { APDC_cabinet_higher_influence_count_socialism < 6 } } set_variable = { APDC_cabinet_top_six_candidate_socialism = 1 } add_to_variable = { APDC_cabinet_top_six_candidate_count = 1 } }

		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_communism = 1 } check_variable = { APDC_cabinet_raw_influence_communism < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_communism } }

		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_liberalism = 1 } check_variable = { APDC_cabinet_raw_influence_liberalism < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_liberalism } }
		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_authoritarianism = 1 } check_variable = { APDC_cabinet_raw_influence_authoritarianism < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_authoritarianism } }
		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_dictatorship = 1 } check_variable = { APDC_cabinet_raw_influence_dictatorship < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_dictatorship } }
		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_fascism = 1 } check_variable = { APDC_cabinet_raw_influence_fascism < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_fascism } }
		if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_fundamentalism = 1 } check_variable = { APDC_cabinet_raw_influence_fundamentalism < APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_top_six_cutoff = APDC_cabinet_raw_influence_fundamentalism } }
		set_variable = { APDC_cabinet_guaranteed_seat_count = 0 }
		if = {
			# Exactly six candidates means there is no contested cutoff.
			limit = { check_variable = { APDC_cabinet_top_six_candidate_count < 7 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_communism = 1 } } set_variable = { APDC_cabinet_available_communism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_socialism = 1 } } set_variable = { APDC_cabinet_available_socialism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_social_democracy = 1 } } set_variable = { APDC_cabinet_available_social_democracy = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_conservatism = 1 } } set_variable = { APDC_cabinet_available_conservatism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_liberalism = 1 } } set_variable = { APDC_cabinet_available_liberalism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_authoritarianism = 1 } } set_variable = { APDC_cabinet_available_authoritarianism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_dictatorship = 1 } } set_variable = { APDC_cabinet_available_dictatorship = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_fascism = 1 } } set_variable = { APDC_cabinet_available_fascism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_fundamentalism = 1 } } set_variable = { APDC_cabinet_available_fundamentalism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
		}
		else = {
			# When the cutoff is tied, only parties strictly above it get a seat.
			# Every contested cutoff seat is forwarded to the next allocation stage.
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_communism = 1 } check_variable = { APDC_cabinet_raw_influence_communism > APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_available_communism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_socialism = 1 } check_variable = { APDC_cabinet_raw_influence_socialism > APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_available_socialism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_social_democracy = 1 } check_variable = { APDC_cabinet_raw_influence_social_democracy > APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_available_social_democracy = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }
			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_conservatism = 1 } check_variable = { APDC_cabinet_raw_influence_conservatism > APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_available_conservatism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }

			if = { limit = { check_variable = { APDC_cabinet_top_six_candidate_authoritarianism = 1 } check_variable = { APDC_cabinet_raw_influence_authoritarianism > APDC_cabinet_top_six_cutoff } } set_variable = { APDC_cabinet_available_authoritarianism = 1 } add_to_variable = { APDC_cabinet_guaranteed_seat_count = 1 } }

		subtract_from_variable = { APDC_cabinet_remaining_seats = APDC_cabinet_guaranteed_seat_count }
	}
	clamp_variable = { var = APDC_cabinet_remaining_seats min = 0 max = 6 }

	# Above-average test without division:
	# party support * coalition-party count > total coalition support.
	set_variable = { APDC_cabinet_above_average_count = 0 }
	set_variable = { APDC_cabinet_above_average_communism = 0 }
	set_variable = { APDC_cabinet_above_average_test_communism = APDC_cabinet_raw_influence_communism }
	multiply_variable = { APDC_cabinet_above_average_test_communism = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_communism_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_communism > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_communism = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_socialism = 0 }
	set_variable = { APDC_cabinet_above_average_test_socialism = APDC_cabinet_raw_influence_socialism }
	multiply_variable = { APDC_cabinet_above_average_test_socialism = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_socialism_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_socialism > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_socialism = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_social_democracy = 0 }
	set_variable = { APDC_cabinet_above_average_test_social_democracy = APDC_cabinet_raw_influence_social_democracy }
	multiply_variable = { APDC_cabinet_above_average_test_social_democracy = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_social_democracy_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_social_democracy > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_social_democracy = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_conservatism = 0 }
	set_variable = { APDC_cabinet_above_average_test_conservatism = APDC_cabinet_raw_influence_conservatism }
	multiply_variable = { APDC_cabinet_above_average_test_conservatism = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_conservatism_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_conservatism > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_conservatism = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_liberalism = 0 }
	set_variable = { APDC_cabinet_above_average_test_liberalism = APDC_cabinet_raw_influence_liberalism }
	multiply_variable = { APDC_cabinet_above_average_test_liberalism = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_liberalism_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_liberalism > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_liberalism = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_authoritarianism = 0 }
	set_variable = { APDC_cabinet_above_average_test_authoritarianism = APDC_cabinet_raw_influence_authoritarianism }
	multiply_variable = { APDC_cabinet_above_average_test_authoritarianism = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_authoritarianism_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_authoritarianism > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_authoritarianism = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_dictatorship = 0 }
	set_variable = { APDC_cabinet_above_average_test_dictatorship = APDC_cabinet_raw_influence_dictatorship }
	multiply_variable = { APDC_cabinet_above_average_test_dictatorship = APDC_cabinet_coalition_party_count }
	if = { limit = { APDC_is_dictatorship_in_governing_coalition = yes check_variable = { APDC_cabinet_above_average_test_dictatorship > APDC_cabinet_coalition_total } } set_variable = { APDC_cabinet_above_average_dictatorship = 1 } add_to_variable = { APDC_cabinet_above_average_count = 1 } }
	set_variable = { APDC_cabinet_above_average_fascism = 0 }
	set_variable = { APDC_cabinet_above_average_test_fascism = APDC_cabinet_raw_influence_fascism }
	multiply_variable = { APDC_cabinet_above_average_test_fascism = APDC_cabinet_coalition_party_count }

	multiply_variable = { APDC_cabinet_above_average_test_fundamentalism = APDC_cabinet_coalition_party_count }

	# Both inputs are small integers, so use exact integer thresholds instead of
	# subtracting 0.5 before round_variable (fixed-point 0.5 ties round to zero).
	set_variable = { APDC_cabinet_extra_per_above_average_party = 0 }
	if = {
		limit = { check_variable = { APDC_cabinet_above_average_count > 0 } }
		set_variable = { APDC_cabinet_extra_seat_threshold = APDC_cabinet_above_average_count }
		if = { limit = { NOT = { check_variable = { APDC_cabinet_remaining_seats < APDC_cabinet_extra_seat_threshold } } } set_variable = { APDC_cabinet_extra_per_above_average_party = 1 } }
		add_to_variable = { APDC_cabinet_extra_seat_threshold = APDC_cabinet_above_average_count }

		add_to_variable = { APDC_cabinet_extra_seat_threshold = APDC_cabinet_above_average_count }

		if = { limit = { NOT = { check_variable = { APDC_cabinet_remaining_seats < APDC_cabinet_extra_seat_threshold } } } set_variable = { APDC_cabinet_extra_per_above_average_party = 6 } }

	if = { limit = { check_variable = { APDC_cabinet_above_average_communism = 1 } } add_to_variable = { APDC_cabinet_available_communism = APDC_cabinet_extra_per_above_average_party } }

	if = { limit = { check_variable = { APDC_cabinet_above_average_social_democracy = 1 } } add_to_variable = { APDC_cabinet_available_social_democracy = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_conservatism = 1 } } add_to_variable = { APDC_cabinet_available_conservatism = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_liberalism = 1 } } add_to_variable = { APDC_cabinet_available_liberalism = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_authoritarianism = 1 } } add_to_variable = { APDC_cabinet_available_authoritarianism = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_dictatorship = 1 } } add_to_variable = { APDC_cabinet_available_dictatorship = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_fascism = 1 } } add_to_variable = { APDC_cabinet_available_fascism = APDC_cabinet_extra_per_above_average_party } }
	if = { limit = { check_variable = { APDC_cabinet_above_average_fundamentalism = 1 } } add_to_variable = { APDC_cabinet_available_fundamentalism = APDC_cabinet_extra_per_above_average_party } }

	set_variable = { APDC_cabinet_distributed_extra_seats = APDC_cabinet_extra_per_above_average_party }
	multiply_variable = { APDC_cabinet_distributed_extra_seats = APDC_cabinet_above_average_count }
	subtract_from_variable = { APDC_cabinet_remaining_seats = APDC_cabinet_distributed_extra_seats }
	clamp_variable = { var = APDC_cabinet_remaining_seats min = 0 max = 6 }

	# Give every final remainder seat to the ruling party.
	if = { limit = { has_government = communism } add_to_variable = { APDC_cabinet_available_communism = APDC_cabinet_remaining_seats } }
	if = { limit = { has_government = socialism } add_to_variable = { APDC_cabinet_available_socialism = APDC_cabinet_remaining_seats } }
